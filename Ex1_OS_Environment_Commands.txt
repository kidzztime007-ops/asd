Exp. No: 1 - Study of Operating System Environment (Linux/Windows commands)
========================================

COMMANDS (LINUX)
----------------------------------------
# Ex.No:1  Linux commands (type these in the terminal, one by one)

# Step 3: system information
uname -a
whoami
pwd
date

# Step 4: directory structure
mkdir oslab
ls -l

# Step 5: file operations
echo "Hello OS Lab" > oslab/file.txt
cat oslab/file.txt
cp oslab/file.txt oslab/copy.txt
wc -l oslab/file.txt

# Step 6: permissions
chmod 600 oslab/file.txt
ls -l oslab

# Step 7: disk and memory
df -h
free -h

# Step 8: process monitoring
ps -ef
top            # press q to quit

# Step 9: remove file
rm oslab/file.txt oslab/copy.txt
ls -l oslab

COMMANDS (WINDOWS)
----------------------------------------
# Ex.No:1  Windows commands (type these in Command Prompt, one by one)

dir
md oslab
echo Hello OS Lab > oslab\file.txt
type oslab\file.txt
copy oslab\file.txt oslab\copy.txt
attrib +r oslab\file.txt
attrib oslab\file.txt
tasklist
attrib -r oslab\file.txt
del oslab\file.txt oslab\copy.txt
dir oslab
